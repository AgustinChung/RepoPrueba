class Juego {
-personajes: List<Personaje>
-criaturas: List<Criaturas>
-misiones: List<MisionSecundaria>

+juego(personajes: List<Personaje>,criaturas: List<Criaturas>): Juego
+agregarPersonaje(personaje: Personaje): void
+agregarCriatura(criatura: Criatura): void
+agregarMision(mision: MisionSecundaria): void
+buscarPersonaje(personaje: Personaje): Personaje
+buscarCriatura(criatura: Criatura): Criatura
+buscarMision(mision: MisionSecundaria): Criatura
}

abstract class Personaje{
-nombre: String
-puntosVida: int
-nivelAtaque: int
-nivelDefensa: int
-nivelExperiencia: int

+nivelDefensaEsCero(nivelDefensa: int): boolean
+usarMapa(mapa: Mapa): void
}


class Guerrero extends Personaje{
-contador: {static} int
-golpesDados: int

+guerrero(nombre: String, puntosVida: int,nivelAtaque: int,nivelDefensa: int,nivelExperiencia: int): Guerrero
+darGolpe(enemigo: Criatura): void
+duplicarAtaque(): void
+viajar(ubicacion: Ubicacion): void
+recolectarRecompensa(mision: MisionSecundaria): void
}

class Mago extends Personaje{

+mago(nombre: String, puntosVida: int,nivelAtaque: int,nivelDefensa: int,nivelExperiencia: int): Mago
+curar(): void
+esInmune(): void
}

class Arquero extends Personaje{
-nivelPunteria: int
-nivelAgilidad: int

+arquero(nombre: String, puntosVida: int,nivelAtaque: int,nivelDefensa: int,nivelExperiencia: int): Arquero
+esquivarGolpe(nivelPunteria: int): void
+aumentarPunteria(nivelPunteria: int, enemigo: Criatura): void

}


abstract class Criatura{
-nombre: String
-puntosVida: int
-nivel: int
-nivelAtaque: int
-nivelDefensa: int

#nivelDefensaEsCero(nivelDefensa: int): boolean
#aumentarAtaque(): void
#aumentarDefensa(): void
#darExperiencia(): void
}

class Dragon extends Criatura{
+dragon(nombre: String,puntosVida: int,nivelAtaque: int,nivelDefensa: int): Dragon
+aumentarAtaque(enemigo: Personaje, nivelAtaque: int): void
+darExperiencia(personaje: Personaje, nivel: int): void
}

class Troll extends Criatura{
+troll(nombre: String,puntosVida: int,nivelAtaque: int,nivelDefensa: int): Troll
+aumentarDefensa(enemigo: Personaje,nivelDefensa: int): void
+darExperiencia(personaje: Personaje, nivel: int): void
}

class Espectro extends Criatura{
+espectro(nombre: String,puntosVida: int,nivelAtaque: int,nivelDefensa: int): Espectro
+aumentarAtaque(enemigo: Personaje, nivelAtaque: int): void
+darExperiencia(personaje: Personaje, nivel: int): void
}

class Pelea{
-personaje: Personaje
-enemigo: Criatura

+iniciarPelea(personaje: Personaje, enemigo: Criatura): Pelea
+danioPersonaje(personaje: Personaje, enemigo: Criatura): void
+danioCriatura(personaje: Personaje, enemigo: Criatura): void
}


class Mapa{
-ubicaciones: List<Ubicacion>

+mostrarUbicaciones(List<Ubicacion>): List <Ubicacion>
+agregarUbicaciones(ubicacion: Ubicacion): void
}


class Ubicacion{
-nombre: String
-tipo: TipoUbicacion
-personaje: Personaje
-criatura: Criatura
-esNeutral: boolean
-tieneTesoro: boolean

+ubicacion(criatura: Criatura, esNeutral: boolean): Ubicacion
+agregarPersonaje(personaje: Personaje): void
+mejorarNivel(personaje: Personaje): void
+descansar(personaje: Personaje): void
+reclamarRecompensa(personaje: Personaje, misionSecundaria: MisionSecundaria): void
}


class MisionSecundaria{
-objetivo: TipoObjetivo
+obtenerRecompensa(objetivo: TipoObjetivo): void
}

enum TipoObjetivo{
DERROTAR DRAGON NORTE
RECUPERAR AMULETO
ELIMINAR ESPECTRO
LIMPIAR ALDEA
}

enum TipoUbicacion{
MONTANIAS HELADAS
BOSQUE DE LOS SUSURROS
PANTANO OSCURO
ALDEA DE LOS SIRITH
}

Juego "1" *-- "0..*" Personaje
Juego "1" *-- "0..*" Criatura
Juego "1" *-- "0..*" MisionSecundaria 


Pelea "1" *-- "1" Personaje
Pelea "1" *-- "1" Criatura

Personaje -> Mapa

Mapa "1" *-- "1..*" Ubicacion

Ubicacion "1" o-- "1" Personaje
Ubicacion "1" o-- "1" Criatura
Ubicacion -> TipoUbicacion

MisionSecundaria -> TipoObjetivo
@enduml
